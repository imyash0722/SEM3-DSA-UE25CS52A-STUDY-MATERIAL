A heap(maxheap) is a tree with two special properties.
1. Complete binary tree:
The nodes are filled up at each level.
At the bottom most level, leftmost nodes are filled up.
2.     Parental dominance:
The key at the root is greater than the keys of its subtrees 
recursively.

static void heapify(int a[], int n, int pos)
{
int i = pos;
int j = 2 * i;
int val = a[i];
int is_heap = 0;
 
while(j <= n && ! is_heap){
if(j + 1 <= n && a[j+1] > a[j])
++j;
if(a[j] > val) {
a[i] = a[j];
i = j;
j = 2 * i;
}
else {
is_heap = 1;
}
 }
a[i] = val;
}

void insert(heap_t *ptr_heap, int key)
{
int pos = ++ptr_heap->n_;
ptr_heap->key_[pos] = key;
while(pos > 1 && ptr_heap->key_[pos] > ptr_heap->key_[pos/2])
{
swap(&ptr_heap->key_[pos], &ptr_heap->key_[pos/2]);
pos = pos / 2;
}
}

Delete max : replace the first node with the last and heapify about the root.
int delete_max(heap_t *ptr_heap)
{
int res = ptr_heap->key_[1];
ptr_heap->key_[1] = ptr_heap->key_[ptr_heap->n_];
heapify(ptr_heap->key_, --ptr_heap->n_, 1);
return res;
}
