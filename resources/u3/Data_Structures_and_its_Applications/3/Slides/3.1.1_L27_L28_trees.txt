Definitiion
• A binary tree is a tree in which every node has at most
two children.
• The two children are distinguished as the
left child and right child.
• A binary tree is either:
• An empty tree, or
• A root node together with
two disjoint binary trees called its:
• left subtree
• right subtree

// on the node; recursive; observe the assignment to the root

static node_t* insert_recursive(node_t* root, node_t* temp)
{
if(root == NULL)
{
root = temp;
}
else if(root->key_ > temp->key_)
{
root->left_ = insert_recursive(root->left_, temp);
}
else
{
root->right_ = insert_recursive(root->right_, temp);
}
return root;
}

int are_same_recursive(node_t* root1, node_t* root2)
{
if(root1 == NULL && root2 == NULL)
return 1;
else if(root1 == NULL || root2 == NULL)
return 0;
else
return root1->key_ == root2->key_ &&
are_same_recursive(root1->left_, root2->left_) &&
are_same_recursive(root1->right_, root2->right_);
}

node_t* copy_recursive(node_t* ptr_src)
{
node_t *temp;
if(ptr_src != NULL)
{
temp = (node_t*)malloc(sizeof(node_t));
temp->key_ = ptr_src->key_;
temp->left_ = copy_recursive(ptr_src->left_);

return temp;

else
{
return NULL;
}

return NULL;
else if(key == root->key_)

else if(key < root->key_)
return search_recursive(root->left_, key);
else

node_t* temp = root;

temp = temp->left_;

Binary Tree : persistence
•
•
•
•
•

A single traversal will not define the tree uniquely
An inorder and a postorder traversal combined define a tree uniquely
Rightmost node in the postorder traversal will be the root of the tree
This node in the order traversal divides the tree in left subtree and the right subtree
Repeating traversal from right to left of post order traversal and locating the node
to divide the inorder traversal to left and right subtrees
will recreate the tree
